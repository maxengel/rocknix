# A stand-in for raofflineproxy-cache-images beside the ctl, driven by
# /run/images-mode: "ok" (exit 0), "fail" (the helper's own failure lines,
# exit 1), "verify-left" (a verify slice left, no why, exit 3), "crash" (a
# traceback and Python's exit 1, no protocol line), "cancel" (the
# page's CANCEL: SIGINT to the ctl through the lock's pid, then a real 20 s),
# "hang" (a real 20 s); every start is logged with its pid and a real time.
dfake_images() {
    cat > "${T}/repo/raofflineproxy-cache-images" <<'EOS'
#!/bin/bash
echo $$ > /run/images.pid
echo "start $(date +%s.%N) budget=${FAKE_BUDGET:-none} $*" >> /run/images-calls
trap 'echo "term $(date +%s.%N)" >> /run/images-calls; exit 143' TERM
case "$(cat /run/images-mode 2>/dev/null || echo ok)" in
    fail) printf '>>> doing images\n>>> total 3\n>>> why SOME_IMAGES_NOT_SAVED\n>>> done 0|3\n'; exit 1 ;;
    verify-left) printf '>>> doing images\n>>> verified 10|40\n>>> note VERIFY_LEFT 30\n>>> total 0\n>>> note NOTHING_MISSING\n>>> done 0|0\n'; exit 3 ;;
    crash) printf '>>> doing images\n'; echo 'Traceback (most recent call last):' >&2; exit 1 ;;
    cancel) kill -INT "$(head -n 1 /var/run/raofflineproxy-scan.lock)"; i=0; while [ "$i" -lt 20 ]; do /bin/sleep 1; i=$((i+1)); done; echo "done $(date +%s.%N)" >> /run/images-calls; exit 0 ;;
    hang) i=0; while [ "$i" -lt 20 ]; do /bin/sleep 1; i=$((i+1)); done; echo "done $(date +%s.%N)" >> /run/images-calls; exit 0 ;;
esac
printf '>>> doing images\n>>> total 0\n>>> note NOTHING_MISSING\n>>> done 0|0\n'
exit 0
EOS
    chmod +x "${T}/repo/raofflineproxy-cache-images"
}
