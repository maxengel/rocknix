p=""; files=0; dirs=0; markers=0; ignore=0; src=""

    case "$a" in
        qa:*) p="$a" ;;
        --files-only) files=1 ;; --dirs-only) dirs=1 ;;
        --s3-directory-markers) markers=1 ;; --ignore-existing) ignore=1 ;;
        /*) src="$a" ;;
    esac
done

  listremotes) echo "qa:" ;;

    mkdir -p "${target}" ;;
  lsf)
    [ -n "${fail}" ] && { echo 'ERROR : Failed to lsf: connection reset' >&2; exit 1; }
    if [ -f "${target}" ]; then basename "${target}"; exit 0; fi
    if [ -d "${target}" ]; then
        for e in "${target}"/*; do
            [ -e "$e" ] || continue
            if [ -d "$e" ]; then [ "${files}" -eq 1 ] || echo "$(basename "$e")/"

        exit 0

    echo 'ERROR : directory not found' >&2; exit 3 ;;

def completed(args, rc, out="", err=""):
    return subprocess.CompletedProcess(args, rc, out, err)

class FakeSP:
    """subprocess, with run() answered by a table: (argv prefix) -> callable."""
    PIPE, STDOUT, DEVNULL = subprocess.PIPE, subprocess.STDOUT, subprocess.DEVNULL
    TimeoutExpired, SubprocessError = subprocess.TimeoutExpired, subprocess.SubprocessError
    CompletedProcess = subprocess.CompletedProcess
    Popen = subprocess.Popen

    def __init__(self, answers):
        self.answers = answers
        self.calls = []

    def run(self, args, **kw):
        self.calls.append(list(args))
        for prefix, answer in self.answers:
            if list(args[1:1 + len(prefix)]) == list(prefix):
                return answer(args)
        return completed(args, 0)

def pid_alive(pid):
    try:
        with open("/proc/%d/stat" % pid) as h:
            return h.read().split(") ")[-1].split()[0] != "Z"
    except OSError:
        return False

def wait_for(pred, seconds, step=0.1):
    end = time.time() + seconds
    while time.time() < end:
        if pred():
            return True
        time.sleep(step)
    return pred()

class FakeFcntl:
    def __init__(self, real, marker):
        self.real = real
        self.marker = marker
        self.grabs = []

    def ioctl(self, fd, req, arg=0, *rest):
        if req == 0x40044590:           # EVIOCGRAB
            self.grabs.append((arg, os.path.exists(self.marker)))
            return 0
        return self.real.ioctl(fd, req, arg, *rest)

def pad_fifo(d):
    fifo = os.path.join(d, "pad")
    os.mkfifo(fifo)
    writer = os.open(fifo, os.O_RDWR)       # keeps a reader's open from blocking

            value = sum(1 << (c - w * 64) for c in codes if w * 64 <= c < (w + 1) * 64)
            out.append("%x" % value)
        return " ".join(out) + "\n"
    with open(os.path.join(dev, "capabilities", "key"), "w") as h:
        h.write(words(key_codes))
    with open(os.path.join(dev, "capabilities", "abs"), "w") as h:
        h.write(words(abs_codes))
    with open(os.path.join(dev, "name"), "w") as h:
        h.write(name + "\n")
    return root

def start_server(m, holder):
    if hasattr(m, "make_server"):
        server = m.make_server("127.0.0.1", 0, holder, lambda: None)
    else:
        import http.server
        server = http.server.ThreadingHTTPServer(("127.0.0.1", 0), m.make_handler(holder, lambda: None))
    threading.Thread(target=server.serve_forever, daemon=True).start()
    return server, server.server_address[1]

def driver_env(d, lan="127.0.0.1"):
    env = dict(os.environ)
    env.update({"COT_DIR": d, "COT_LAN": lan})
    return env

def drive(d, *args, lan="127.0.0.1", wait=True, **kw):
    cmd = [sys.executable, os.path.abspath(__file__), "drive", REPO, WORK] + list(args)
    p = subprocess.Popen(cmd, env=driver_env(d, lan), stdout=subprocess.PIPE,
                         stderr=subprocess.PIPE, text=True, **kw)
    if wait:
        out, err = p.communicate(timeout=60)
        return p.returncode, out, err
    return p

def state_of(d):
    try:
        with open(os.path.join(d, "state", "session.json")) as h:
            return real_json.load(h)
    except (OSError, ValueError):
        return {}

def procs_with(text):
    """Live processes whose command line, joined with spaces, holds text."""
    found = []
    for e in os.listdir("/proc"):
        if not e.isdigit() or int(e) == os.getpid():
            continue
        try:
            with open("/proc/%s/cmdline" % e, "rb") as h:
                line = h.read().replace(b"\0", b" ").decode("utf-8", "replace")
        except OSError:
            continue
        if text in line and pid_alive(int(e)):
            found.append(int(e))
    return found

def serve_dir(tag):
    d = tempfile.mkdtemp(dir=WORK, prefix=tag + ".")
    os.makedirs(os.path.join(d, "state"))
    return d

function attrs(text) {
  const out = {};

# Each of webkitgtk's patches says what it fixes before its first hunk --
# 0001 was a bare hunk (#308 claude F-RS-26) -- and every one still applies
# to the pinned release, dry, as scripts/unpack applies it: `patch` reads
# the prose before the first --- as a comment. The release's tarball comes
# from the sources cache; without it the dry run is skipped and says so.
WKP="${ROOT}/packages/web/webkitgtk/patches"
for wp in "${WKP}"/*.patch; do
    [ -n "$(sed -n '/^--- /q; /[^[:space:]]/p' "${wp}")" ]; check $? "$(basename "${wp}") says what it fixes before its first hunk" "$(basename "${wp}") is a bare hunk: nothing says which build error it fixes, or why it is carried (#308 claude F-RS-26)"
done
WKV="$(sed -n 's/^PKG_VERSION="\(.*\)"$/\1/p' "${ROOT}/packages/web/webkitgtk/package.mk" | head -1)"
WKT=""
for d in "${SOURCES_DIR:-}" /workspace/cache/rocknix-sources "${ROOT}/sources" "${ROOT}"/../*/sources; do
    [ -n "${d}" ] && [ -f "${d}/webkitgtk/webkitgtk-${WKV}.tar.xz" ] && { WKT="${d}/webkitgtk/webkitgtk-${WKV}.tar.xz"; break; }
done
if [ -z "${WKT}" ]; then
    echo "    SKIP  webkitgtk-${WKV}.tar.xz is not under SOURCES_DIR, /workspace/cache/rocknix-sources or a build worktree's sources/ -- the patches' dry run needs the release they apply to"
else
    WKD="${CS}/wk"; mkdir -p "${WKD}"
    wkfiles=$(sed -n 's|^+++ b/\([^[:space:]]*\).*|webkitgtk-'"${WKV}"'/\1|p' "${WKP}"/*.patch | sort -u)
    # shellcheck disable=SC2086
    tar -xJf "${WKT}" -C "${WKD}" ${wkfiles} 2>/dev/null
    for wp in "${WKP}"/*.patch; do
        patch -d "${WKD}/webkitgtk-${WKV}" -p1 --dry-run < "${wp}" > "${WKD}/out" 2>&1; check $? "$(basename "${wp}") applies to webkitgtk ${WKV}" "$(basename "${wp}") does not apply to ${WKV}: $(tail -2 "${WKD}/out" | tr '\n' ' ')"
    done
fi

# C8. cloud-signin-window.c. The scripts it runs in the page are C string
# literals; they are pulled out as the compiler would join them and run by
# node against a small document. The C around them is compiled only by the
# image build (and syntax-checked with the image's toolchain by whoever
# changes it).
C8="${CS}/c8"; mkdir -p "${C8}"
src_of projects/ROCKNIX/packages/network/cloud-signin-window/sources/cloud-signin-window.c "${C8}/window.c"
cat > "${C8}/cstrings.py" <<'EOPY'
# Print the C string a `static const char *NAME =` declaration builds from its
# adjacent literals, in the file given -- the JavaScript the window runs.
import re, sys
src = open(sys.argv[1]).read()
m = re.search(r"static const char \*%s =(.*?);\n" % re.escape(sys.argv[2]), src, re.S)
if not m:
    sys.exit("no %s in %s" % (sys.argv[2], sys.argv[1]))
body = re.sub(r"/\*.*?\*/", "", m.group(1), flags=re.S)
out = "".join(bytes(s, "utf-8").decode("unicode_escape") for s in re.findall(r'"((?:[^"\\]|\\.)*)"', body))
sys.stdout.write(out)
EOPY
cstr() { python3 "${C8}/cstrings.py" "${C8}/window.c" "$1" 2>/dev/null; }

const fs = require("fs");
const src = fs.readFileSync(process.argv[2], "utf8");
function run(active) {
  let focused = [];
  const field = { tagName: "INPUT", offsetParent: {}, disabled: false, focus() { focused.push("field"); } };
  const body = { tagName: "BODY" };
  const document = { body, documentElement: { tagName: "HTML" },
    activeElement: active === "body" ? body : active === "button" ? { tagName: "BUTTON" } : field,
    querySelector: () => field };

}

# The page file (the address the window shows, for the phone to notice a
# change) is not written for rclone's loopback redirect, whose query carries
# the OAuth code (#308 claude F-RS-20). page_worth_recording is plain C, so
# it is lifted out and compiled with the host's cc against the addresses.
sed -n '/^static int page_worth_recording(/,/^}/p' "${C8}/window.c" > "${C8}/pwr.c"
cat > "${C8}/pwr_main.c" <<'EOC'
#include <stdio.h>
#include <string.h>
#include <strings.h>
#include "pwr.c"
int main(void)
{
    const char *keep[] = { "https://www.dropbox.com/oauth2/authorize?state=x",
                           "https://login.live.com/ppsecure", "https://127.0.0.1.example.com/", 0 };
    const char *skip[] = { "http://127.0.0.1:53682/?state=abc&code=SECRET",
                           "http://localhost:53682/?code=SECRET", "http://LOCALHOST/", 0 };
    int bad = 0;
    for (int i = 0; keep[i]; i++) if (!page_worth_recording(keep[i])) { printf("dropped %s\n", keep[i]); bad++; }
    for (int i = 0; skip[i]; i++) if (page_worth_recording(skip[i])) { printf("recorded %s\n", skip[i]); bad++; }
    return bad;
}
EOC
C8P=$( [ -s "${C8}/pwr.c" ] && command -v cc >/dev/null && cc -std=gnu99 -o "${C8}/pwr" "${C8}/pwr_main.c" 2>&1 && "${C8}/pwr" 2>&1 )
[ -s "${C8}/pwr.c" ] && [ -x "${C8}/pwr" ] && "${C8}/pwr" >/dev/null 2>&1; check $? "the page file is not written for rclone's loopback redirect, which carries the code; a provider's own pages still are" "page_worth_recording: $([ -s "${C8}/pwr.c" ] || echo 'no such function in cloud-signin-window.c -- every URI, the loopback redirect with its code included, goes to the page file'; echo "${C8P}" | head -3 | tr '\n' ' ') (#308 claude F-RS-20)"

# The on-screen keyboard reaches every printable character a password may
# hold: the two layouts lacked = : ; ' " , < > [ ] { } \ | ` ~ (#308 gpt
# F-RS-12). Read from the layout tables themselves: every layout's rows are
# as long as the lower-case one's (the keys are built from it, and a shorter
# row would leave the last layout's labels on the rest), and the extras row
# holds the key that reaches the symbols.
cat > "${C8}/osk.py" <<'EOPY'
import re, sys
src = open(sys.argv[1]).read()
def table(name):
    m = re.search(r"static const char \*%s\[[^\]]*\] = \{(.*?)\};" % name, src, re.S)
    if not m:
        return None
    return [bytes(t, "utf-8").decode("unicode_escape") for t in re.findall(r'"((?:[^"\\]|\\.)*)"', m.group(1))]
lower, extras = table("OSK_LOWER"), table("OSK_EXTRAS") or []
layouts = [t for t in (table(n) for n in ("OSK_LOWER", "OSK_UPPER", "OSK_SYMBOLS")) if t]
chars = set("".join("".join(t) for t in layouts)) | {e for e in extras if len(e) == 1}
if "space" in extras:
    chars.add(" ")
missing = "".join(sorted(set(chr(c) for c in range(32, 127)) - chars))
ragged = [n for n, t in zip(("OSK_UPPER", "OSK_SYMBOLS"), layouts[1:]) if [len(r) for r in t] != [len(r) for r in lower]]
print("missing=%s|ragged=%s" % (missing, ",".join(ragged)))
EOPY
C8K=$(python3 "${C8}/osk.py" "${C8}/window.c" 2>&1)
[ "${C8K}" = "missing=|ragged=" ]; check $? "the on-screen keyboard reaches every printable ASCII character, each layout's rows as long as the keys built for them" "the on-screen keyboard: ${C8K} -- a password with one of those cannot be typed on the handheld (#308 gpt F-RS-12)"
