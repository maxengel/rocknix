# A directory's files: the working tree's, or BASE_REF's under --old, or a
# named ref's (f1_tree_at) -- an absent directory copies nothing.
f1_tree_at() { # <ref|""> <repo-relative dir> <destination>
    mkdir -p "${3}"
    if [ -n "${1}" ]; then
        git -C "${ROOT}" ls-tree --name-only "${1}:${2}" 2>/dev/null | while IFS= read -r f; do
            git -C "${ROOT}" show "${1}:${2}/${f}" > "${3}/${f}" || { echo "cannot read ${2}/${f} at ${1}" >&2; exit 2; }
        done
    elif [ -d "${ROOT}/${2}" ]; then
        cp -a "${ROOT}/${2}/." "${3}/"
    fi
    chmod +x "${3}"/* 2>/dev/null; true
}
f1_tree() { if [ "${OLD}" -eq 1 ]; then f1_tree_at "${BASE_REF}" "$@"; else f1_tree_at "" "$@"; fi; }

    unmask) shift; for u in "$@"; do [ "$(readlink "/storage/.config/system.d/${u}")" = /dev/null ] && rm -f "/storage/.config/system.d/${u}"; done ;;
esac
exit 0
EOS

# /storage as a guest's first boot leaves it: the /usr/config directories the
# links point into (each with its README), the shipped retroarch.cfg.
f1_storage() { # <dir>: a fresh /storage
    rm -rf "${1}"; mkdir -p "${1}/.config/profile.d" "${1}/.config/retroarch" "${1}/.cache"
    for d in tmpfiles.d sysctl.d modules-load.d modprobe.d udev.rules.d hwdb.d system.d logind.conf.d resolved.conf.d sleep.conf.d timesyncd.conf.d; do
        mkdir -p "${1}/.config/${d}"; echo "shipped" > "${1}/.config/${d}/README"
    done
    printf 'video_driver = "vulkan"\nvideo_fullscreen = "true"\nvideo_fullscreen_x = "640"\nvideo_fullscreen_y = "480"\n' > "${1}/.config/retroarch/retroarch.cfg"
}
# /sys: one connected virtio output at 1280x800 (QEMU's default), fw_cfg's
# SSH port, the DMI names.
f1_sys() { # <dir> [<connector>:<status>:<mode>[,<mode>...]]...
    local d="${1}" c name st modes; shift
    rm -rf "${d}"; mkdir -p "${d}/class/drm" "${d}/firmware/qemu_fw_cfg/by_name/opt/org.rocknix.cloud_ssh_port" "${d}/class/dmi/id"
    printf '10022' > "${d}/firmware/qemu_fw_cfg/by_name/opt/org.rocknix.cloud_ssh_port/raw"
    echo QEMU > "${d}/class/dmi/id/sys_vendor"; echo "Standard PC (Q35 + ICH9, 2009)" > "${d}/class/dmi/id/product_name"
    [ $# -eq 0 ] && set -- "card0-Virtual-1:connected:1280x800,1024x768,640x480"
    for c in "$@"; do
        name="${c%%:*}"; c="${c#*:}"; st="${c%%:*}"; modes="${c#*:}"
        mkdir -p "${d}/class/drm/${name}"
        echo "${st}" > "${d}/class/drm/${name}/status"
        if [ "${st}" = connected ]; then echo enabled; else echo disabled; fi > "${d}/class/drm/${name}/enabled"
        echo "${modes}" | tr ',' '\n' > "${d}/class/drm/${name}/modes"
    done
}
# One boot's quirk pass, as autostart runs it: the platform's scripts, then
# the device's, each "${script}" 2>&1 >>boot.log -- so what a script says on
# stderr is what the journal holds, and that is what lands in <journal>.
f1_pass() { # <storage> <sys> <quirks dir> <journal>
    ( bwrap --die-with-parent --tmpfs / --ro-bind /usr /usr "${F1BB[@]}" \
        --symlink usr/bin /bin --symlink usr/lib /lib --symlink usr/lib64 /lib64 --symlink usr/sbin /sbin \
        --ro-bind "${F1ETC}" /etc --bind "${1}" /storage --ro-bind "${2}" /sys --ro-bind "${3}" /quirks \
        --ro-bind "${F1}/shim" /shim --tmpfs /run --tmpfs /tmp --tmpfs /var --dev /dev --proc /proc \
        --setenv PATH "${F1PATH}" --setenv HOME /storage \
        bash -c 'for s in /quirks/platforms/GENERIC_X64/* "/quirks/devices/QEMU Standard PC (Q35 + ICH9, 2009)"/*; do
                     [ -f "${s}" ] && bash "${s}" 2>&1 >>/run/boot.log
                 done; exit 0' ) > "${4}" 2>&1
}
f1_quirks() { # <ref|""> <destination>: the platform and QEMU device quirks a build installs

    f1_tree_at "${1}" "${F1_QREL}/platforms/GENERIC_X64" "${2}/platforms/GENERIC_X64"
    f1_tree_at "${1}" "${F1_QREL}/devices/${F1_QEMU}" "${2}/devices/${F1_QEMU}"
}

# F1e. #307 PL-073: the in-place bootloader updater. The SYSTEM image's
#      /usr/share/bootloader holds update.sh alone (projects/ROCKNIX/
#      bootloader/install copies the syslinux files after the squashfs is
#      made, into the release's 3rdparty/bootloader), so every guarded copy

#      /storage/.boot.hint, which init writes itself after

#      script is gone.

        --ro-bind "${F1E}/update" /update --bind "${F1E}/flash" /flash --bind "${F1E}/storage" /storage --ro-bind "${F1E}/shim" /shim \
