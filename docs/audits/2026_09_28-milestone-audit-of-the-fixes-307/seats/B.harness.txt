p7bt() { # <verb...>: backuptool whole; stdout+stderr to $P7OUT, rc in $RC
    RC=$( ( setsid -w bwrap --die-with-parent --tmpfs / "${P7UB[@]}" --ro-bind "${P7U}" /usr/config "${P7BB[@]}" \
        --symlink usr/bin /bin --symlink usr/lib /lib --symlink usr/lib64 /lib64 \
        --ro-bind /etc /etc --ro-bind "${P7}/profile" /etc/profile \
        --bind "${P7S}" /storage --ro-bind "${P7}/shim" /p7shim --ro-bind "${P7}/bb" /p7bb --ro-bind "${P7}/repo" /repo \
        "${P7X[@]}" --dev /dev --proc /proc --tmpfs /tmp --chdir /storage/.p7cwd \
        bash /repo/backuptool "$@" > "${P7OUT}" 2>&1; echo $? ) 2>/dev/null )
}
p7reset() { # a device with a system.cfg carrying a Wi-Fi key and the given backuptool.conf body (none when empty)
    chmod -R u+rwX "${P7S}" 2>/dev/null; rm -rf "${P7S}"
    mkdir -p "${P7S}/.config/system/configs" "${P7S}/.config/test" "${P7S}/.config/emulationstation" "${P7S}/roms/backup" "${P7S}/.p7cwd"
    printf 'system.hostname=X\nwifi.key=SECRET\n' > "${P7S}/.config/system/configs/system.cfg"
    echo file1 > "${P7S}/.config/test/f1"
    [ -n "${1:-}" ] && printf 'LOCATIONS=(\n%s\n)\n' "$1" > "${P7S}/.config/backuptool.conf"
    P7X=()
    return 0
}
p7arc() { ls "${P7S}/roms/backup"/*.tar.gz 2>/dev/null; }
p7list() { tar -tzf "$(p7arc | tail -1)" 2>/dev/null; }
p7why() { grep '^>>> why ' "${P7OUT}" | tail -1 | cut -c9-; }
p7screen() { grep -vE '^logger |^>>> ' "${P7OUT}" | tr '\n' '|' | cut -c1-240; }

p7tree() { # <dir> <relpath=content>...: a fixture tree
    local d="$1" pair p; shift; rm -rf "${d}"; mkdir -p "${d}"
    for pair in "$@"; do p="${pair%%=*}"; mkdir -p "${d}/$(dirname "${p}")"; printf '%s\n' "${pair#*=}" > "${d}/${p}"; done
}
p7tar() { # <archive name under roms/backup> <relpath=content>...: a tar archive made by the host's tar
    local n="$1"; shift; p7tree "${P7}/tree" "$@"; tar -C "${P7}/tree" -czf "${P7S}/roms/backup/${n}" storage
}
p7zip() { # <archive name under roms/backup> <relpath=content>...: a zip archive, the shape before 2026-08-26
    local n="$1"; shift; p7tree "${P7}/tree" "$@"
    (cd "${P7}/tree" && python3 -c 'import zipfile, os, sys
z = zipfile.ZipFile(sys.argv[1], "w", zipfile.ZIP_DEFLATED)
for r, _, fs in os.walk("storage"):
    for f in sorted(fs): z.write(os.path.join(r, f))
z.close()' "${P7S}/roms/backup/${n}")
}
# A tar that extracts for real and then fails -- the restore's extraction cut
# part-way; every other tar call (the snapshot's own extraction included) is
# the image's. It keeps a copy of the marker as the extraction found it.
p7failtar() {
    cat > "${P7}/shim/tar" <<'EOS'
#!/bin/bash
if [ "$1" = "-xzf" ] && [[ "$2" != */archive/* ]] && [ "$3" = "-C" ] && [ "$4" = "/" ]; then
    cp /storage/.config/.restore-in-progress /storage/.p7-marker-seen 2>/dev/null
    /p7bb/tar "$@"
    echo "tar: can't open 'storage/.config/test/f1': No space left on device" >&2
    exit 1
fi
exec /p7bb/tar "$@"
EOS
    chmod +x "${P7}/shim/tar"
}

p7ck() { # <stdin producer> <exe> <epoch> [env...]: the keeper under the image's head; the note of <exe>.<epoch> in $P7NOTE

        --ro-bind /etc /etc --bind "${P7C}/storage" /storage --ro-bind "${P7C}/shim" /shim --ro-bind "${P7C}/bb" /p7bb --ro-bind "${P7C}/repo" /repo "${P7MBB[@]}" \
        --dev /dev --proc /proc --tmpfs /tmp --tmpfs /var --setenv PATH /shim:/p7bb:/usr/bin:/bin "$@" \
        bash -c "${prod} | bash /repo/rocknix-corekeep ${exe} 4242 ${when} 11 '!usr!bin!${P7EXEPATH:-${exe}}'" ) > "${P7C}/out" 2>&1
    P7NOTE="$(cat "${P7C}"/storage/.cache/log/cores/core.*."${when}".4242.txt 2>/dev/null)"
}

import subprocess, sys
p = subprocess.Popen(sys.argv[2], shell=True, stdin=subprocess.PIPE, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
chunk = b"\0" * 65536; sent = 0
try:
    for _ in range(1024):              # 64 MiB offered
        p.stdin.write(chunk); p.stdin.flush(); sent += len(chunk)
except (BrokenPipeError, OSError):
    pass
try: p.stdin.close()
except Exception: pass
p.wait(); print(sent)
EOS
)
