# F2-5..F2-13: #308's rows in the recipes, one case each. A recipe is
# sourced as source_package sources it -- PKG_DIR set, ROOT the tree's root
# (a fixture root where the case builds one) -- in a subshell with set +u,
# and its hooks are called in the order scripts/unpack and scripts/build
# call them; nothing is compiled, fetched or installed.
f2_source() { # <recipe file> <DEVICE> <ARCH> [ROOT] -- then run "$F2_CALL" in the recipe's shell
    ( set +u; DEVICE="$2"; ARCH="$3"; TARGET_ARCH="$3"; ROOT="${4:-${ROOT}}"; PKG_DIR="$(dirname "$1")"
      OPENGL=mesa; OPENGLES=no; DISPLAYSERVER=wl; SYSROOT_PREFIX=/sysroot; TARGET_NAME=t; TARGET_PREFIX=t-
      die() { echo "DIE: $*"; exit 3; }
      safe_remove() { local p; for p in "$@"; do [ -e "${p}" ] || [ -L "${p}" ] && rm -r "${p}"; done; }
      get_build_dir() { echo "/build/$1"; }
      get_pkg_directory() { echo "${ROOT}/packages/${1}"; }
      . "$1" >/dev/null 2>&1
      eval "${F2_CALL}" )
}
